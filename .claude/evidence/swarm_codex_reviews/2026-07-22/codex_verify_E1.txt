  490: def sharpe_ratio_stderr(
  491:     returns: np.ndarray | list[float], adjust_autocorr: bool = True
  492: ) -> float:
  493:     """Standard error of a per-period Sharpe estimate.
  494: 
  495:     With ``adjust_autocorr`` the SE is inflated for first-order serial correlation
  496:     (Lo 2002). Overlapping windows and trend-following returns are positively
  497:     autocorrelated, and ignoring that makes every confidence interval too narrow.
  498:     """
  499:     import math
  500: 
  501:     r = np.asarray(returns, dtype=float)
  502:     r = r[np.isfinite(r)]
  503:     n = len(r)
  504:     if n < 3:
  505:         return float("inf")
  506:     sd = r.std(ddof=1)
  507:     if sd <= 0:
  508:         return float("inf")
  509:     sr = r.mean() / sd
  510:     var = (1.0 + 0.5 * sr * sr) / (n - 1)
  511:     se = math.sqrt(var)
  512:     if adjust_autocorr and n > 3:
  513:         rho = float(np.corrcoef(r[:-1], r[1:])[0, 1])
  514:         if np.isfinite(rho) and abs(rho) < 0.999:
  515:             se *= math.sqrt((1.0 + rho) / (1.0 - rho))
  516:     return float(se)

  519: def purged_kfold_splits(t1, n_splits: int = 5, embargo_pct: float = 0.01):
  520:     """K-fold splits with purging AND embargo (LÃ³pez de Prado, AFML ch. 7).
  521: 
  522:     Plain K-fold leaks whenever label windows straddle a fold boundary: the training set
  523:     then contains information about the test period and OOS Sharpe comes back inflated.
  524:     Every multi-day-hold strategy in this repo has that exposure.
  525: 
  526:     Purging drops training samples whose label window overlaps the test window; the embargo
  527:     additionally drops samples immediately AFTER the test window, where serial correlation
  528:     still carries test information.
  529: 
  530:     Args:
  531:         t1: ``pd.Series`` of label END times, indexed by label start time, sorted ascending.
  532:         n_splits: number of folds (>= 2).
  533:         embargo_pct: fraction of the sample embargoed after each test fold.
  534: 
  535:     Yields:
  536:         ``(train_indices, test_indices)`` positional integer arrays.
  537:     """
  538:     import pandas as pd
  539: 
  540:     if not isinstance(t1, pd.Series):
  541:         raise TypeError("t1 must be a pd.Series of label end times")
  542:     if not t1.index.is_monotonic_increasing:
  543:         raise ValueError("t1 index must be sorted ascending")
  544:     if n_splits < 2:
  545:         raise ValueError("n_splits must be >= 2")
  546:     if not 0.0 <= embargo_pct < 1.0:
  547:         raise ValueError("embargo_pct must be in [0, 1)")
  548: 
  549:     n = len(t1)
  550:     indices = np.arange(n)
  551:     embargo = int(n * embargo_pct)
  552: 
  553:     for test_idx in np.array_split(indices, n_splits):
  554:         test_start_time = t1.index[test_idx[0]]
  555:         test_end_time = t1.iloc[test_idx].max()
  556:         before = indices[(t1.values < test_start_time)]
  557:         after_pos = int(t1.index.searchsorted(test_end_time, side="right"))
  558:         after = indices[min(after_pos + embargo, n):]
  559:         train_idx = np.concatenate([before, after])
  560:         if train_idx.size == 0:
  561:             raise ValueError(
  562:                 "Purging removed the entire training set for a fold. Label horizon is too "
  563:                 f"long relative to sample length (n={n}, n_splits={n_splits})."
  564:             )
  565:         yield train_idx, test_idx

  568: def pbo_cscv(returns_matrix, n_blocks: int = 16) -> dict:
  569:     """Probability of Backtest Overfitting via Combinatorially Symmetric CV.
  570: 
  571:     DSR deflates a Sharpe for how many trials you ran. PBO asks the complementary and
  572:     arguably harsher question: **when I pick the best variant in-sample, does it stay above
  573:     median out-of-sample?** PBO is the fraction of splits where it does not.
  574: 
  575:     ``pbo > 0.5`` means the selection procedure is worse than random â€” the in-sample winner
  576:     tends to be an out-of-sample loser. That is a REJECT regardless of Sharpe.
  577: 
  578:     Args:
  579:         returns_matrix: ``(T observations x N strategies)``, N >= 2.
  580:         n_blocks: even, <= 20. C(n_blocks, n_blocks/2) splits are evaluated.
  581: 
  582:     Returns:
  583:         ``{pbo, n_combinations, n_strategies}``.
  584:     """
  585:     import math
  586:     from itertools import combinations
  587: 
  588:     from scipy import stats
  589: 
  590:     m = np.asarray(returns_matrix, dtype=float)
  591:     if m.ndim != 2:
  592:         raise ValueError("returns_matrix must be 2-D (T x N)")
  593:     t, n_strat = m.shape
  594:     if n_strat < 2:
  595:         raise ValueError("PBO needs >= 2 strategies to rank")
  596:     if n_blocks % 2 != 0:
  597:         raise ValueError("n_blocks must be even")
  598:     if n_blocks > 20:
  599:         raise ValueError(f"n_blocks={n_blocks} gives too many combinations")
  600:     if t < n_blocks * 2:
  601:         raise ValueError(f"need >= {n_blocks * 2} observations for {n_blocks} blocks")
  602: 
  603:     blocks = np.array_split(np.arange(t), n_blocks)
  604:     logits: list[float] = []
  605:     for combo in combinations(range(n_blocks), n_blocks // 2):
  606:         is_idx = np.concatenate([blocks[i] for i in combo])
  607:         oos_idx = np.concatenate([blocks[i] for i in range(n_blocks) if i not in combo])
  608:         with np.errstate(invalid="ignore", divide="ignore"):
  609:             sd_is = m[is_idx].std(axis=0, ddof=1)
  610:             sd_oos = m[oos_idx].std(axis=0, ddof=1)
  611:             sr_is = np.where(sd_is > 0, m[is_idx].mean(axis=0) / sd_is, -np.inf)
  612:             sr_oos = np.where(sd_oos > 0, m[oos_idx].mean(axis=0) / sd_oos, -np.inf)
  613:         if not np.isfinite(sr_is).any():
  614:             continue
  615:         best = int(np.nanargmax(sr_is))
  616:         rank = float(stats.rankdata(sr_oos)[best]) / (n_strat + 1)
  617:         logits.append(math.log(rank / (1.0 - rank)))
  618: 
  619:     if not logits:
  620:         raise ValueError("no valid splits - check for degenerate return series")
  621:     arr = np.asarray(logits)
  622:     return {
  623:         "pbo": round(float((arr <= 0).mean()), 4),
  624:         "n_combinations": int(arr.size),
  625:         "n_strategies": int(n_strat),
  626:     }

  633: def omega_ratio(returns, threshold: float = 0.0) -> float | None:
  634:     """Omega = sum(gains above threshold) / sum(losses below it).
  635: 
  636:     Uses the WHOLE distribution -- skew and kurtosis included -- where Sharpe uses only its
  637:     first two moments. That matters here because the edge in this system is risk control, and
  638:     risk control shows up as asymmetry that a Sharpe cannot see.
  639: 
  640:     Returns None when there are no losses, never inf (JSON safety, strategy-contract rule 2).
  641:     """
  642:     r = np.asarray(returns, dtype=float)
  643:     r = r[~np.isnan(r)]
  644:     if r.size == 0:
  645:         return None
  646:     gains = np.sum(np.clip(r - threshold, 0, None))
  647:     losses = np.sum(np.clip(threshold - r, 0, None))
  648:     if losses <= 0:
  649:         return None
  650:     return float(round(gains / losses, 4))

  653: def capture_ratios(returns, benchmark) -> dict:
  654:     """Up/down capture against a benchmark, and their ratio.
  655: 
  656:     The target profile is up_capture > 100 and down_capture < 100: participate in the rises,
  657:     sit out the falls. This is the honest way to frame a strategy whose directional forecast
  658:     has no edge -- it asks "what share of each side did you take", not "did you predict".
  659:     """
  660:     r = np.asarray(returns, dtype=float)
  661:     b = np.asarray(benchmark, dtype=float)
  662:     n = min(len(r), len(b))
  663:     r, b = r[:n], b[:n]
  664:     ok = ~(np.isnan(r) | np.isnan(b))
  665:     r, b = r[ok], b[ok]
  666: 
  667:     out: dict = {}
  668:     for name, mask in (("up", b > 0), ("down", b < 0)):
  669:         if not mask.any() or np.isclose(b[mask].mean(), 0):
  670:             out[f"{name}_capture_pct"] = None
  671:             continue
  672:         out[f"{name}_capture_pct"] = float(round(r[mask].mean() / b[mask].mean() * 100, 2))
  673:     up, dn = out.get("up_capture_pct"), out.get("down_capture_pct")
  674:     # A negative or zero down-capture means the strategy GAINED while the benchmark fell; the
  675:     # ratio is not meaningful there, so it is None rather than a large flattering number.
  676:     out["capture_ratio"] = float(round(up / dn, 3)) if (up and dn and dn > 0) else None
  677:     return out

  680: def dsr_report(sharpe_per_period: float, n_obs: int, n_trials: int, *,
  681:                variant_sharpes=None, sigma_grid=(0.05, 0.10, 0.15),
  682:                periods_per_year: int = 252, skew: float = 0.0,
  683:                kurtosis: float = 3.0) -> dict:
  684:     """Deflated Sharpe reported over EVERY sigma assumption, headline = the least flattering.
  685: 
  686:     Exists because the same strategy was getting two different DSRs on 2026-07-21:
  687:     `gold_trend_simple` scored 0.861 in the evidence harness (sigma estimated from the
  688:     dispersion of published variants) and 0.0033 in its publisher (sigma hardcoded to
  689:     `0.5/sqrt(252)`). Neither was a bug -- they were different assumptions -- but a number that
  690:     swings 250x on an undocumented constant is a number someone will eventually quote
  691:     selectively. There is one DSR entry point now, and it always shows its work.
  692: 
  693:     UNITS, which is where this goes wrong: `deflated_sharpe_ratio` needs
  694:     `trials_sharpe_std` in the SAME units as `sharpe_per_period`. Registries declare sigma
  695:     annualized, so a declared 0.10 becomes 0.10/sqrt(periods_per_year) here. Feeding an
  696:     annualized sigma against a per-period Sharpe makes the null expect random strategies to
  697:     achieve annualized Sharpes of 2-6, under which nothing can ever pass.
  698: 
  699:     `variant_sharpes` (per-period Sharpes of the asset's actually-published variants) gives the
  700:     EMPIRICAL estimate, which needs no unit conversion because it is measured in the right
  701:     units by construction. It is preferred when >= 3 variants exist; the declared grid is the
  702:     fallback and is always published alongside so the sensitivity is visible.
  703:     """
  704:     root = float(np.sqrt(periods_per_year))
  705:     cells, sigmas = [], []
  706: 
  707:     if variant_sharpes is not None:
  708:         vs = np.asarray([s for s in variant_sharpes if s is not None and np.isfinite(s)],
  709:                         dtype=float)
  710:         if vs.size >= 3:
  711:             sigmas.append((float(np.std(vs, ddof=1)), "empirical_from_published_variants"))
  712: 
  713:     sigmas += [(float(s) / root, f"declared_{s}_annualized_over_sqrt({periods_per_year})")
  714:                for s in sigma_grid]
  715: 
  716:     for sig, origin in sigmas:
  717:         if sig <= 0:
  718:             continue
  719:         d = deflated_sharpe_ratio(sharpe_per_period, n_obs, n_trials, sig,
  720:                                   skew=skew, kurtosis=kurtosis)
  721:         cells.append({"sigma_trials_pp": round(sig, 6), "sigma_origin": origin, **d})
  722: 
  723:     if not cells:
  724:         return {"headline_dsr": None, "cells": [], "bar": 0.95, "passes": False,
  725:                 "reason": "no usable sigma"}
  726: 
  727:     preferred = [c for c in cells if c["sigma_origin"] == "empirical_from_published_variants"]
  728:     headline = min(preferred or cells, key=lambda c: c["dsr"])
  729:     return {
  730:         "headline_dsr": headline["dsr"],
  731:         "headline_sigma_origin": headline["sigma_origin"],
  732:         "headline_sr0": headline["sr0"],
  733:         "worst_cell_dsr": min(c["dsr"] for c in cells),
  734:         "n_trials": int(n_trials), "n_obs": int(n_obs),
  735:         "sharpe_per_period": round(float(sharpe_per_period), 6),
  736:         "cells": cells, "bar": 0.95, "passes": bool(headline["dsr"] > 0.95),
  737:     }

  740: def ewma_volatility(returns, lam: float = 0.94, periods_per_year: int = 365,
  741:                     seed_window: int = 20) -> np.ndarray:
  742:     """RiskMetrics EWMA volatility, annualized. sigma^2_t = lam*sigma^2_{t-1} + (1-lam)*r^2_{t-1}.
  743: 
  744:     Skill: vendor/quant-skills/04-backtesting-validation/volatility-modeling.
  745: 
  746:     Why this over a rolling window: an equal-weighted 20-day window treats a shock from 20 days
  747:     ago exactly like yesterday's, then drops it off a cliff when it leaves the window. EWMA
  748:     decays smoothly. Effective window is ~1/(1-lam) = 17 days at lam=0.94, the RiskMetrics daily
  749:     standard.
  750: 
  751:     Honest caveat the skill itself raises: EWMA is IGARCH (alpha+beta=1), so vol shocks persist
  752:     INDEFINITELY with no mean reversion. For an asset whose volatility mean-reverts hard, that
  753:     can keep position size suppressed long after the shock has passed. This is a real reason it
  754:     may perform WORSE than a rolling window, not a footnote.
  755: 
  756:     Strictly causal: sigma_t uses returns up to t-1 only, so it can size the bar at t.
  757:     """
  758:     r = np.asarray(returns, dtype=float)
  759:     r = np.nan_to_num(r, nan=0.0)
  760:     n = len(r)
  761:     if n == 0:
  762:         return np.zeros(0)
  763: 
  764:     out = np.empty(n)
  765:     seed = r[:min(seed_window, n)]
  766:     var = float(np.var(seed, ddof=0)) if seed.size else 0.0
  767:     if var <= 0:
  768:         var = float(np.var(r, ddof=0)) or 1e-8
  769:     for i in range(n):
  770:         out[i] = np.sqrt(var * periods_per_year)   # sigma_t, built from r_{<t}
  771:         var = lam * var + (1.0 - lam) * (r[i] ** 2)
  772:     return out

  775: def qlike_loss(true_var, pred_var) -> np.ndarray:
  776:     """QLIKE loss per observation: true/pred - ln(true/pred) - 1.
  777: 
  778:     The standard loss for variance forecasts (Patton 2011: one of only two losses robust to
  779:     noisy volatility proxies; the other is MSE on variance). RMSE on sigma rewards
  780:     under-prediction because vol is right-skewed -- a forecaster that always guesses low looks
  781:     good on RMSE and then under-sizes the one week that matters. QLIKE penalizes
  782:     under-prediction asymmetrically, which is the direction risk actually hurts.
  783: 
  784:     Element-wise; observations where either side is non-positive return NaN rather than being
  785:     silently dropped -- the caller decides what a zero-variance day means for its series.
  786:     """
  787:     t = np.asarray(true_var, dtype=float)
  788:     p = np.asarray(pred_var, dtype=float)
  789:     out = np.full(np.broadcast(t, p).shape, np.nan)
  790:     ok = (t > 0) & (p > 0)
  791:     ratio = t[ok] / p[ok]
  792:     out[ok] = ratio - np.log(ratio) - 1.0
  793:     return out

  796: def circular_block_bootstrap(x, stat_fn, n_boot: int = 2000, block: int = 4,
  797:                              seed: int = 42) -> dict:
  798:     """Circular block bootstrap of an arbitrary statistic over a (possibly paired) series.
  799: 
  800:     Panel audit 2026-07-21: every analysis was re-implementing ad-hoc bootstraps; this is
  801:     the SSOT. `block=4` weekly preserves the monthly dependence structure that iid
  802:     resampling destroys. For PAIRED comparisons pass the delta series and a stat_fn over
  803:     it -- pairing must happen before resampling, never after.
  804: 
  805:     Returns {stat, ci95, se, n_boot, block}.
  806:     """
  807:     rng = np.random.default_rng(seed)
  808:     x = np.asarray(x, dtype=float)
  809:     x = x[np.isfinite(x)]
  810:     n = len(x)
  811:     if n < block + 1:
  812:         return {"stat": None, "ci95": [None, None], "se": None,
  813:                 "n_boot": 0, "block": block, "reason": f"n={n} < block+1"}
  814:     stats = np.empty(n_boot)
  815:     n_blocks = int(np.ceil(n / block))
  816:     for b in range(n_boot):
  817:         starts = rng.integers(0, n, size=n_blocks)
  818:         idx = (starts[:, None] + np.arange(block)[None, :]).ravel() % n
  819:         stats[b] = stat_fn(x[idx[:n]])
  820:     return {"stat": float(stat_fn(x)),
  821:             "ci95": [float(np.percentile(stats, 2.5)), float(np.percentile(stats, 97.5))],
  822:             "se": float(np.std(stats, ddof=1)), "n_boot": n_boot, "block": block}
