  167: def epoch_ms_to_month(ms: int) -> date:
  168:     """SUAMECA stamps = Bogota-midnight of the month's last day (UTC-5).
  169: 
  170:     Converting the instant to UTC yields e.g. 2000-01-31T05:00Z, whose UTC
  171:     calendar date is still inside the reference month, so year/month of the
  172:     UTC date identify the month unambiguously. Instant-based, no COT localize.
  173:     """
  174:     ts = datetime.fromtimestamp(ms / 1000.0, tz=timezone.utc)
  175:     return date(ts.year, ts.month, 1)

  179:     """Conservative PIT date: last calendar day of month+1."""
  180:     y, m = (month.year, month.month + 1) if month.month < 12 else (month.year + 1, 1)
  181:     return date(y, m, calendar.monthrange(y, m)[1])

  206:     y, m = start.year, start.month

  212: def validate(rows: list[dict]) -> None:
  213:     if not rows:
  214:         raise SystemExit("[FAIL] no rows parsed")

  216:     # 1) no negatives
  217:     negatives = [r for r in rows if r["remesas_usd_mn"] < 0]
